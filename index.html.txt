        // Tab 切換 Logic
        function switchTab(tabId, btn) {
            document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
            document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active'));
            
            document.getElementById(tabId).classList.add('active');
            btn.classList.add('active');
            window.scrollTo(0, 0);
        }

        // 自動儲存與載入筆記
        const notesArea = document.getElementById('travelNotes');
        if (notesArea) {
            notesArea.value = localStorage.getItem('nagoya_travel_notes') || '';
            notesArea.addEventListener('input', () => {
                localStorage.setItem('nagoya_travel_notes', notesArea.value);
            });
        }

        // 自動儲存伴手禮勾選狀態
        document.querySelectorAll('.checklist-item input[type="checkbox"]').forEach((chk, index) => {
            const savedState = localStorage.getItem('souvenir_' + index);
            if (savedState === 'true') chk.checked = true;
            chk.addEventListener('change', () => {
                localStorage.setItem('souvenir_' + index, chk.checked);
            });
        });
    </script>
